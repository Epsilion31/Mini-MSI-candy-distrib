#include <Arduino.h>
#include <Wire.h>
#include <Servo.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64

#define PIN_BOUTON 4
#define PIN_SERVO  5

Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, -1);
Servo pushServo;

void afficherEtat(int etat) {
  display.clearDisplay();
  display.setTextSize(2);
  display.setTextColor(SSD1306_WHITE);
  
  display.setCursor(20, 20);
  display.print("Etat: ");
  display.print(etat);
  
  display.display();
}

void setup() {
  pinMode(PIN_BOUTON, INPUT);

  pushServo.attach(PIN_SERVO);
  pushServo.write(0); // Position repos

  display.begin(SSD1306_SWITCHCAPVCC, 0x3C);
  afficherEtat(0); // Affichage état 0 au démarrage
}

void loop() {
  // Si on appuie sur le bouton
  if (digitalRead(PIN_BOUTON) == HIGH) {
    // 1. Passage à l'état 1
    afficherEtat(1);

    // 2. Le moteur tourne
    pushServo.write(90); // Pousse
    delay(500);
    pushServo.write(0);  // Revient
    delay(500);

    // 3. Retour à l'état 0
    afficherEtat(0);

    // Pause anti-rebond
    delay(500);
  }
}
